import biosense as bs
from biosense.dsp import welch_psd, band_power

# Initialize 8-ch EEG stream over LSL socket
stream = bs.DeviceStream(modality='EEG', srate=1000)
raw = stream.get_eeg(channels=['Cz', 'F3', 'F4'], filter_band=(0.5, 45.0))

# Compute band integral for Alpha focus index
psd, freqs = welch_psd(raw['Cz'], nperseg=2048, overlap=0.5)
alpha_pow = band_power(psd, freqs, band=[8.0, 12.0])
print(f"[BioSense] Cz Alpha Power: {alpha_pow:.2f} µV²")
