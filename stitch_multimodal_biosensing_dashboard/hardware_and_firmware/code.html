AFE DMA Initialization

// ESP-IDF v5.1 SPI Bus & DMA Ring Buffer Configuration
#include "driver/spi_master.h"
#include "esp_timer.h"

static spi_device_handle_t eeg_spi_handle;

void init_ads1299_dma(void) {
    spi_bus_config_t buscfg = {
        .mosi_io_num = 11,
        .miso_io_num = 13,
        .sclk_io_num = 12,
        .quadwp_io_num = -1,
        .quadhd_io_num = -1,
        .max_transfer_sz = 4096
    };
    spi_bus_initialize(SPI2_HOST, &buscfg, SPI_DMA_CH_AUTO);

    spi_device_interface_config_t devcfg = {
        .clock_speed_hz = 4 * 1000 * 1000, // 4 MHz
        .mode = 1,                         // CPOL=0, CPHA=1
        .spics_io_num = 10,
        .queue_size = 16
    };
    spi_bus_add_device(SPI2_HOST, &devcfg, &eeg_spi_handle);
}
